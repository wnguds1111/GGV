js_Login() 함수 — 핵심 로그인 로직
function js_Login(game, pw, recapaStr) {
    var inputData = {
        gnId: game,                          // GNJOY 계정 ID
        password: pw,                        // 비밀번호
        recaptcha: recapaStr,                // Cloudflare Turnstile 토큰
        __RequestVerificationToken:          // ASP.NET AntiForgery 토큰
            $('input[name="__RequestVerificationToken"]').val()
    };
    
    $.ajax({
        cache: false,
        type: 'POST',
        dataType: 'json',
        url: "../Login/Login",              // POST /Login/Login
        data: inputData,
        success: function(result) {
            turnstile_reset('#cf_recapa0');  // Turnstile 위젯 리셋
            
            let go_url = window.parent.location.href;
            
            // ✅ 성공: statusCode == "1"
            if (result.statusCode == "1") {
                // Member, RO, LoginFail 페이지에서 로그인하면 → Member/Logined로
                if (go_url.indexOf('/Member/') > -1 ||
                    go_url.indexOf('/RO/') > -1 ||
                    go_url.indexOf('/LoginFail') > -1) {
                    window.parent.location.href = "../Member/Logined";
                    return;
                }
                window.parent.location.reload();  // 다른 페이지에서는 새로고침
                return;
            }
            
            // ❌ 정지 계정
            if (result.responseText == '帳號已被停權') {
                window.parent.location.href = "../Login/StopStates";
                return;
            }
            
            // ❌ 기타 에러 → SweetAlert 표시
            MsgBox(result.responseText);
        },
        error: function(xhr) {
            if (xhr.status == 400) {
                swal('請刷新頁面');  // "페이지 새로고침" (CSRF 토큰 만료)
            }
            turnstile_reset('#cf_recapa0');
        }
    });
};

// 헤더의 "登出" 클릭 시 실행
$('.signout').click(function() {
    event.preventDefault();
    var inputData = {
        gnId: $('#Lgameid').val(),
        __RequestVerificationToken: $('input[name="__RequestVerificationToken"]').val()
    }
    $.ajax({
        cache: false,
        type: 'POST',
        url: '/Login/Logout',       // POST /Login/Logout
        dataType: 'json',
        data: inputData,
        success: function(result) {
            if (result.statusCode == "0") {
                MsgBox(result.responseText);
                return;
            }
            // Login/Member/RO 페이지 → Signin으로
            // 기타 페이지 → 현재 페이지 reload
            if (controller == 'login' || controller == 'member' || controller == 'ro') {
                window.parent.location.href = "../Member/Signin";
            } else {
                window.parent.location.reload();
            }
        }
    });
});

LoginChk & MemberLoginedChk — 네비게이션 함수
// 서버에서 CurrentUser 값을 함수 인자로 주입
// 비로그인 시 빈 문자열 → 리다이렉트 트리거

function LoginChk(input) {
    if (input == '') {
        MsgBox("請先登入平台帳號才能使用本功能");
        setTimeout(() => window.location.href = '../Billing', 1500);
        // 비로그인 → 1.5초 후 Billing (로그인 유도)
    } else {
        window.location.href = '../Billing';
        // 로그인 → 바로 Billing 이동
    }
}

function MemberLoginedChk(input) {
    if (input == '') {
        window.location.href = '../Member/Signin';
        // 비로그인 → 로그인 페이지
    } else {
        window.location.href = '../Member/Logined';
        // 로그인 → 회원 정보 페이지
    }
}

// 네비게이션에서의 호출:
// onclick="MemberLoginedChk('')"  ← 비로그인 시 빈 문자열
// onclick="LoginChk('')"          ← 비로그인 시 빈 문자열
